\chapter{Esercizi — Equazioni di Secondo Grado (fino alle equazioni monomie)}

\section{Ripasso: equazioni di primo grado}

\begin{mdframed}
Porta tutti i termini con la $x$ da una parte dell'uguale e i numeri dall'altra, poi dividi per il coefficiente di $x$.
\end{mdframed}

\begin{exercise}[\onestar]
Risolvi l'equazione:
$$
-\frac34 x + 2 = \frac{x}{2} - 1
$$
\end{exercise}

\begin{exercise}[\onestar]
Risolvi l'equazione:
$$
3(x-2) - \frac{x+1}{3} = x-5
$$
\end{exercise}

\section{Quante soluzioni ha?}

\begin{mdframed}
Isola $x^2$: se il risultato è negativo l'equazione non ha soluzioni, se è zero ne ha una, se è positivo ne ha due.
\end{mdframed}

\begin{exercise}[\onestar]
Quante soluzioni ha l'equazione
$$
3x^2-27=0 \, ?
$$
\begin{sceltamultipla}
\item $0$
\item $1$
\item $2$
\end{sceltamultipla}
\end{exercise}

\begin{exercise}[\onestar]
Quante soluzioni ha l'equazione
$$
-2x^2=0 \, ?
$$
\begin{sceltamultipla}
\item $0$
\item $1$
\item $2$
\end{sceltamultipla}
\end{exercise}

\section{Forma normale: individua $a$, $b$, $c$}

\begin{mdframed}
Porta tutti i termini a sinistra dell'uguale, riduci i termini simili e confronta con $ax^2+bx+c=0$.
\end{mdframed}

\begin{exercise}[\onestar]
$$
3x^2-5+2x=x^2+3x-1
$$
\end{exercise}

\begin{exercise}[\onestar]
$$
-\frac{x^2}{4}+3=0
$$
\end{exercise}

\begin{exercise}[\onestar]
$$
-3x = 2x^2+1
$$
\end{exercise}

\begin{exercise}[\threestars]
Osserva l'equazione
$$
(2x+1)^2 = 4x^2+4x+5
$$
\begin{truthorfalse}
\tfitem{È un'equazione di secondo grado}
\end{truthorfalse}
\end{exercise}

\section{Equazioni monomie: risolvi}

\begin{mdframed}
Riduci l'equazione alla forma $ax^2=0$: l'unica soluzione è sempre $x=0$.
\end{mdframed}

\begin{exercise}[\onestar]
$$
-4x^2=0
$$
\end{exercise}

\begin{exercise}[\onestar]
$$
5x^2 - \frac{x^2}{2} = 0
$$
\end{exercise}

\begin{exercise}[\twostars]
\textit{Attenzione, occhio al trabocchetto!}
$$
3(x+1)^2 = 6x+3
$$
\end{exercise}

\begin{exercise}[\threestars]
Per quale valore del parametro $k$ l'equazione
$$
2x^2+kx=5x
$$
è un'equazione monomia? In quel caso, qual è la sua soluzione?
\end{exercise}

\section{Vero o falso: è una soluzione?}

\begin{mdframed}
Sostituisci il valore indicato al posto della $x$ e verifica se l'uguaglianza ottenuta è vera.
\end{mdframed}

\begin{exercise}[\onestar]
\begin{truthorfalse}
\tfitem{$x=3$ è soluzione di $x^2-5x+6=0$}
\tfitem{$x=-1$ è soluzione di $2x^2+3x-1=0$}
\tfitem{$x=0$ è soluzione di $5x^2-2x=0$}
\tfitem{$x=1$ è soluzione di $(x-1)^2=0$}
\tfitem{$x=1$ è soluzione di $3x^2-12=0$}
\end{truthorfalse}
\end{exercise}

\chapter{Soluzioni}

\begin{expsolution}[1]
$x=\dfrac{12}{5}$.
\end{expsolution}

\begin{expsolution}[2]
$x=\dfrac45$.
\end{expsolution}

\begin{expsolution}[3]
$x^2=9 \Rightarrow$ 2 soluzioni ($x=\pm3$).
\end{expsolution}

\begin{expsolution}[4]
$x^2=0 \Rightarrow$ 1 sola soluzione ($x=0$).
\end{expsolution}

\begin{expsolution}[5]
$2x^2-x-4=0$: \quad $a=2$, $b=-1$, $c=-4$.
\end{expsolution}

\begin{expsolution}[6]
Già in forma normale: \quad $a=-\dfrac14$, $b=0$, $c=3$.
\end{expsolution}

\begin{expsolution}[7]
$-2x^2-3x-1=0$: \quad $a=-2$, $b=-3$, $c=-1$.
\end{expsolution}

\begin{expsolution}[8]
Sviluppando il quadrato: $4x^2+4x+1=4x^2+4x+5$, cioè $1=5$: un'uguaglianza impossibile, in cui anche il termine di secondo grado si elide. L'affermazione è \textbf{Falsa}: non è un'equazione di secondo grado (anzi, non ha alcuna soluzione).
\end{expsolution}

\begin{expsolution}[9]
$x=0$.
\end{expsolution}

\begin{expsolution}[10]
$\dfrac92 x^2=0 \Rightarrow x=0$.
\end{expsolution}

\begin{expsolution}[11]
Sviluppando: $3x^2+6x+3=6x+3 \Rightarrow 3x^2=0 \Rightarrow x=0$.
\end{expsolution}

\begin{expsolution}[12]
$2x^2+(k-5)x=0$ è monomia quando $k-5=0$, cioè $k=5$. In quel caso l'equazione è $2x^2=0$, con soluzione $x=0$.
\end{expsolution}

\end{expsolution}
